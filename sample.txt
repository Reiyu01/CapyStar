import httpx
from fastapi import FastAPI, Request, HTTPException
from fastapi.responses import StreamingResponse, JSONResponse

app = FastAPI(title="AI Gateway MVP")

# MVP 階段先硬寫，之後換成 DB
VALID_KEYS = {"sk-test-key-001"}

# A 機器的 vLLM 端點
VLLM_BASE = "http://140.133.77.129:8000"


def verify_key(request: Request) -> str:
    """從 x-api-key header 或 Authorization: Bearer 取得並驗證 key"""
    key = request.headers.get("x-api-key", "")
    if not key:
        auth = request.headers.get("Authorization", "")
        key = auth.removeprefix("Bearer ").strip()
    if key not in VALID_KEYS:
        raise HTTPException(status_code=403, detail="Invalid or missing API key")
    return key


@app.post("/v1/chat/completions")
async def chat_completions(request: Request):
    verify_key(request)
    body = await request.json()
    is_stream = body.get("stream", False)
    target = f"{VLLM_BASE}/v1/chat/completions"

    async with httpx.AsyncClient(timeout=120) as client:
        if is_stream:
            async def event_stream():
                async with client.stream("POST", target, json=body) as resp:
                    async for chunk in resp.aiter_text():
                        yield chunk

            return StreamingResponse(event_stream(), media_type="text/event-stream")

        else:
            resp = await client.post(target, json=body)
            return JSONResponse(content=resp.json(), status_code=resp.status_code)


@app.get("/health")
async def health():
    return {"status": "ok", "upstream": VLLM_BASE}
